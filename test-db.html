Database Schema / Permission Note:
If Table Read or Table Write indicates PGRST205 (table not found in cache) or permission error, run the following SQL script in your Supabase SQL Editor: 
-- 1. Create table if not present
create table if not exists public.pickups (
  id bigint generated always as identity primary key,
  booking_ref text not null,
  full_name text not null,
  phone text not null,
  address text not null,
  service text not null,
  estimated_weight text,
  pickup_date text,
  pickup_time text,
  notes text,
  status text default 'pending',
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Grant table permissions to anon & authenticated
grant usage on schema public to anon, authenticated;
grant all on table public.pickups to anon, authenticated;

-- 3. Enable RLS and setup all CRUD policies for anonymous clients
alter table public.pickups enable row level security;

create policy "Allow anon insert" on public.pickups for insert to anon with check (true);
create policy "Allow anon select" on public.pickups for select to anon using (true);
create policy "Allow anon update" on public.pickups for update to anon using (true);
create policy "Allow anon delete" on public.pickups for delete to anon using (true);

-- 4. Reload PostgREST schema cache
notify pgrst, 'reload schema';
